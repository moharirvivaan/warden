cmake_minimum_required(VERSION 3.25)
project(WardenCompatibilityProof LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
if(NOT APPLE OR NOT CMAKE_OSX_ARCHITECTURES STREQUAL "arm64")
  message(FATAL_ERROR "The controlled proof supports macOS ARM64 only")
endif()
if(NOT DEFINED ENV{GEODE_SDK})
  message(FATAL_ERROR "Run scripts/configure-proof.py with the pinned cached SDK")
endif()
add_subdirectory($ENV{GEODE_SDK} "${CMAKE_BINARY_DIR}/geode")

function(proof_mod target folder mod_id)
  add_library(${target} SHARED "${folder}/src/main.cpp")
  target_link_libraries(${target} PRIVATE geode-sdk)
  target_compile_definitions(${target} PRIVATE GEODE_MOD_ID="${mod_id}")
  set_target_properties(${target} PROPERTIES PREFIX "" OUTPUT_NAME "${mod_id}"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/binaries")
endfunction()
proof_mod(proof_remover remover warden.proof-remover)
proof_mod(proof_consumer consumer warden.proof-consumer)
proof_mod(proof_consumer_guarded consumer-guarded warden.proof-consumer-guarded)
